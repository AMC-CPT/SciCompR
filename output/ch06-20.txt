function (x) 
{
    .Scalar1("erf", x)
    if (is.nan(x)) 
        return(NaN)
    if (x == 0) 
        return(0)
    if (x < 0) 
        return(-erf(-x))
    if (x > 27) 
        return(1)
    return(gammp(0.5, x * x))
}
function (x) 
{
    .Scalar1("erfc", x)
    if (is.nan(x)) 
        return(NaN)
    if (x == 0) 
        return(1)
    if (x < 0) 
        return(2 - erfc(-x))
    return(gammq(0.5, x * x))
}
